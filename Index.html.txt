<script>
    let audioCtx, analyser, timerStart = null;

    function toggleCheat() {
        const card = document.getElementById('cheat-card');
        card.style.display = card.style.display === 'block' ? 'none' : 'block';
    }

    async function startApp() {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        analyser = audioCtx.createAnalyser();
        const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
        audioCtx.createMediaStreamSource(stream).connect(analyser);
        draw();
    }

    function draw() {
        requestAnimationFrame(draw);
        const data = new Uint8Array(analyser.frequencyBinCount);
        analyser.getByteFrequencyData(data);
        const canvas = document.getElementById('visualizer');
        const ctx = canvas.getContext('2d');
        const slice = ctx.getImageData(1, 0, canvas.width-1, canvas.height);
        ctx.putImageData(slice, 0, 0);
        
        let maxVal = 0, maxIdx = 0;
        for(let i=0; i<data.length; i++) {
            const v = data[i];
            const y = canvas.height - (i/data.length * canvas.height * 5);
            ctx.fillStyle = `rgb(${v}, ${v/4}, 180)`;
            ctx.fillRect(canvas.width-1, y, 1, 1);
            if(v > maxVal && i < 200) { maxVal = v; maxIdx = i; }
        }

        const hz = Math.round(maxIdx * audioCtx.sampleRate / analyser.fftSize);
        if(maxVal > 50) {
            document.getElementById('p-now').innerText = hz + "Hz";
            const isPitch = hz > 180; const isRes = hz > 200;
            document.getElementById('r-now').innerText = isRes ? "Bright" : "Dark";
            
            if(isPitch) document.getElementById('b1').classList.add('unlocked');
            if(isRes) document.getElementById('b2').classList.add('unlocked');
            
            if(isPitch && isRes) {
                document.getElementById('b3').classList.add('unlocked');
                if(!timerStart) timerStart = Date.now();
                const diff = Math.floor((Date.now() - timerStart)/1000);
                document.getElementById('t-now').innerText = diff + "s";
                if(diff >= 60) document.getElementById('b4').classList.add('unlocked');
            } else {
                timerStart = null;
                document.getElementById('t-now').innerText = "0s";
            }
        }
    }
</script>
